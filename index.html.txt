    <div class="container">
        <div class="card">
            <h2>Привет, <span id="user-name">друг</span>! 👋</h2>

        </div>

        <div class="card">
            <h3>📝 Что хочешь сделать?</h3>
            <select id="action-select">
                <option value="info">Узнать информацию</option>
                <option value="calc">Калькулятор</option>
                <option value="joke">Получить шутку</option>
            </select>
            <input type="text" id="user-input" placeholder="Напиши что-нибудь...">
            <button id="send-btn">Отправить в бот</button>
        </div>

        <div class="card" id="result-card" style="display: none;">
            <h3>📨 Ответ:</h3>
            <p id="result-text"></p>
        </div>
    </div>

    <script>
        // Получаем объект Telegram WebApp
        const tg = window.Telegram.WebApp;
        
        // Говорим Telegram, что приложение готово
        tg.ready();
        
        // Растягиваем на весь экран
        tg.expand();
        
        // Показываем основную кнопку
        tg.MainButton.setText("Готово");
        tg.MainButton.show();
        
        // Данные пользователя (если есть)
        const user = tg.initDataUnsafe?.user;
        if (user) {
            document.getElementById('user-name').innerText = user.first_name || user.username || 'друг';
        }
        
        // Меняем тему под Telegram
        if (tg.colorScheme === 'dark') {
            document.body.style.background = '#1a1a2e';
        }
        
        // Основная кнопка
        tg.MainButton.onClick(() => {
            tg.sendData(JSON.stringify({
                action: 'main_button',
                message: 'Пользователь нажал "Готово"'
            }));
            tg.close();
        });
        
        // Отправка данных в бот
        document.getElementById('send-btn').onclick = () => {
            const action = document.getElementById('action-select').value;
            const userText = document.getElementById('user-input').value;
            
            const data = {
                action: action,
                text: userText,
                user_id: user?.id,
                username: user?.username
            };
            
            // Отправляем данные боту
            tg.sendData(JSON.stringify(data));
            
            // Показываем результат
            const resultCard = document.getElementById('result-card');
            const resultText = document.getElementById('result-text');
            resultText.innerText = `Отправлено! Бот скоро ответит.\nДействие: ${action}\nТекст: ${userText || '(пусто)'}`;
            resultCard.style.display = 'block';
        };
    </script>
